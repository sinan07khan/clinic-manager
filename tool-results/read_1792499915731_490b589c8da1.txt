   267→   267→function StaffDirectory() {
   268→   268→  const [staff, setStaff] = useState<StaffMember[]>([])
   269→   269→  const [loading, setLoading] = useState(true)
   270→   270→  const [search, setSearch] = useState("")
   271→   271→  const [filterRole, setFilterRole] = useState("")
   272→   272→  const [filterDept, setFilterDept] = useState("")
   273→   273→  const [filterActive, setFilterActive] = useState("")
   274→   274→
   275→   275→  // Dialog states
   276→   276→  const [showAddDialog, setShowAddDialog] = useState(false)
   277→   277→  const [editingStaff, setEditingStaff] = useState<StaffMember | null>(null)
   278→   278→
   279→   279→  // Form state
   280→   280→  const [formName, setFormName] = useState("")
   281→   281→  const [formEmail, setFormEmail] = useState("")
   282→   282→  const [formPhone, setFormPhone] = useState("")
   283→   283→  const [formRole, setFormRole] = useState("STAFF")
   284→   284→  const [formDept, setFormDept] = useState("")
   285→   285→  const [formSpec, setFormSpec] = useState("")
   286→   286→  const [formActive, setFormActive] = useState(true)
   287→   287→  const [submitting, setSubmitting] = useState(false)
   288→   288→
   289→   289→  const fetchStaff = useCallback(async () => {
   290→   290→    setLoading(true)
   291→   291→    try {
   292→   292→      const params = new URLSearchParams()
   293→   293→      if (search) params.set("search", search)
   294→   294→      if (filterRole) params.set("role", filterRole)
   295→   295→      if (filterDept) params.set("department", filterDept)
   296→   296→      if (filterActive) params.set("isActive", filterActive)
   297→   297→
   298→   298→      const res = await fetch(`/api/staff?${params.toString()}`)
   299→   299→      if (res.ok) {
   300→   300→        const data = await res.json()
   301→   301→        setStaff(data)
   302→   302→      }
   303→   303→    } catch (err) {
   304→   304→      console.error(err)
   305→   305→    } finally {
   306→   306→      setLoading(false)
   307→   307→    }
   308→   308→  }, [search, filterRole, filterDept, filterActive])
   309→   309→
   310→   310→  useEffect(() => {
   311→   311→    fetchStaff()
   312→   312→  }, [fetchStaff])
   313→   313→
   314→   314→  const resetForm = () => {
   315→   315→    setFormName("")
   316→   316→    setFormEmail("")
   317→   317→    setFormPhone("")
   318→   318→    setFormRole("STAFF")
   319→   319→    setFormDept("")
   320→   320→    setFormSpec("")
   321→   321→    setFormActive(true)
   322→   322→    setEditingStaff(null)
   323→   323→  }
   324→   324→
   325→   325→  const openAdd = () => {
   326→   326→    resetForm()
   327→   327→    setShowAddDialog(true)
   328→   328→  }
   329→   329→
   330→   330→  const openEdit = (s: StaffMember) => {
   331→   331→    setEditingStaff(s)
   332→   332→    setFormName(s.name)
   333→   333→    setFormEmail(s.email)
   334→   334→    setFormPhone(s.phone)
   335→   335→    setFormRole(s.role)
   336→   336→    setFormDept(s.department || "")
   337→   337→    setFormSpec(s.specialization || "")
   338→   338→    setFormActive(s.isActive)
   339→   339→    setShowAddDialog(true)
   340→   340→  }
   341→   341→
   342→   342→  const handleSubmit = async () => {
   343→   343→    if (!formName || !formEmail || !formPhone) {
   344→   344→      toast.error("Name, email, and phone are required")
   345→   345→      return
   346→   346→    }
   347→   347→    setSubmitting(true)
   348→   348→    try {
   349→   349→      const url = editingStaff ? `/api/staff/${editingStaff.id}` : "/api/staff"
   350→   350→      const method = editingStaff ? "PUT" : "POST"
   351→   351→      const body = {
   352→   352→        name: formName,
   353→   353→        email: formEmail,
   354→   354→        phone: formPhone,
   355→   355→        role: formRole,
   356→   356→        department: formDept || null,
   357→   357→        specialization: formSpec || null,
   358→   358→        isActive: formActive,
   359→   359→      }
   360→   360→
   361→   361→      const res = await fetch(url, {
   362→   362→        method,
   363→   363→        headers: { "Content-Type": "application/json" },
   364→   364→        body: JSON.stringify(body),
   365→   365→      })
   366→   366→
   367→   367→      if (res.ok) {
   368→   368→        toast.success(editingStaff ? "Staff updated successfully" : "Staff added successfully")
   369→   369→        setShowAddDialog(false)
   370→   370→        resetForm()
   371→   371→        fetchStaff()
   372→   372→      } else {
   373→   373→        const data = await res.json()
   374→   374→        toast.error(data.error || "Operation failed")
   375→   375→      }
   376→   376→    } catch (err) {
   377→   377→      toast.error("An error occurred")
   378→   378→    } finally {
   379→   379→      setSubmitting(false)
   380→   380→    }
   381→   381→  }
   382→   382→
   383→   383→  const handleToggleActive = async (s: StaffMember) => {
   384→   384→    try {
   385→   385→      const res = await fetch(`/api/staff/${s.id}`, {
   386→   386→        method: "PUT",
   387→   387→        headers: { "Content-Type": "application/json" },
   388→   388→        body: JSON.stringify({ isActive: !s.isActive }),
   389→   389→      })
   390→   390→      if (res.ok) {
   391→   391→        toast.success(`Staff ${s.isActive ? "deactivated" : "activated"}`)
   392→   392→        fetchStaff()
   393→   393→      }
   394→   394→    } catch (err) {
   395→   395→      toast.error("Failed to update staff status")
   396→   396→    }
   397→   397→  }

   402→   402→      const res = await fetch(`/api/staff/${s.id}`, { method: "DELETE" })

   404→   404→        toast.success("Staff deactivated")
   405→   405→        fetchStaff()
   406→   406→      }
   407→   407→    } catch (err) {
   408→   408→      toast.error("Failed to deactivate staff")
   409→   409→    }
   410→   410→  }
   411→   411→
   412→   412→  // Role count summary
   413→   413→  const roleCounts = staff.reduce<Record<string, number>>((acc, s) => {
   414→   414→    acc[s.role] = (acc[s.role] || 0) + 1

   418→   418→  const activeCount = staff.filter((s) => s.isActive).length
   419→   419→  const inactiveCount = staff.length - activeCount
   420→   420→
   421→   421→  return (
   422→   422→    <div className="space-y-6">
   423→   423→      {/* Summary Cards */}
   424→   424→      <div className="grid grid-cols-2 gap-4 sm:grid-cols-3 lg:grid-cols-5">
   425→   425→        <Card className="border-green-200 bg-green-50/50">
   426→   426→          <CardContent className="p-4">
   427→   427→            <div className="flex items-center gap-2">
   428→   428→              <Users className="h-4 w-4 text-green-700" />
   429→   429→              <span className="text-sm text-green-800">Total Staff</span>
   430→   430→            </div>
   431→   431→            <p className="mt-1 text-2xl font-bold text-green-950">{staff.length}</p>
   432→   432→          </CardContent>
   433→   433→        </Card>
   434→   434→        <Card className="border-green-200 bg-green-50/50">
   435→   435→          <CardContent className="p-4">
   436→   436→            <div className="flex items-center gap-2">
   437→   437→              <UserCheck className="h-4 w-4 text-green-600" />
   438→   438→              <span className="text-sm text-green-700">Active</span>
   439→   439→            </div>
   440→   440→            <p className="mt-1 text-2xl font-bold text-green-900">{activeCount}</p>
   441→   441→          </CardContent>
   442→   442→        </Card>
   443→   443→        <Card className="border-red-200 bg-red-50/50">
   444→   444→          <CardContent className="p-4">
   445→   445→            <div className="flex items-center gap-2">
   446→   446→              <UserX className="h-4 w-4 text-red-600" />

   451→   451→        </Card>
   452→   452→        <Card className="border-green-200 bg-green-50/50">

   472→   472→      {/* Filters & Actions */}
   473→   473→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">

   477→   477→            <Input
   478→   478→              placeholder="Search by name..."

   480→   480→              onChange={(e) => setSearch(e.target.value)}
