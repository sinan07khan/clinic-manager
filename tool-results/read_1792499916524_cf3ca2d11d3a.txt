   106→   106→function getStatusColor(status: string) {
   107→   107→  switch (status) {
   108→   108→    case 'BOOKED': return 'bg-blue-100 text-blue-800 border-blue-200'
   109→   109→    case 'CONFIRMED': return 'bg-yellow-100 text-yellow-800 border-yellow-200'

   113→   113→    default: return 'bg-gray-100 text-gray-700 border-gray-200'
   114→   114→  }
