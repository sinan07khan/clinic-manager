    87→    87→function getPOStatusBadge(status: string) {

   111→   111→function StockLevelBar({ quantity, minStock }: { quantity: number; minStock: number }) {
   112→   112→  const percentage = Math.min((quantity / (minStock * 2)) * 100, 100)
   113→   113→  const isLow = quantity < minStock
   114→   114→  const isCritical = quantity === 0
   115→   115→
   116→   116→  return (
   117→   117→    <div className="w-full">
   118→   118→      <div className="h-2 w-full bg-muted rounded-full overflow-hidden">
   119→   119→        <div
   120→   120→          className={`h-full rounded-full transition-all ${
   121→   121→            isCritical ? "bg-red-500" : isLow ? "bg-orange-400" : "bg-green-600"
   122→   122→          }`}
   123→   123→          style={{ width: `${percentage}%` }}
   124→   124→        />
   125→   125→      </div>
   126→   126→      <div className="flex justify-between mt-1">
   127→   127→        <span className={`text-xs ${isCritical ? "text-red-600 font-bold" : isLow ? "text-orange-600 font-medium" : "text-green-700"}`}>
   128→   128→          {quantity} {isLow ? "(Low!)" : ""}
   129→   129→        </span>
   130→   130→        <span className="text-xs text-muted-foreground">Min: {minStock}</span>
   131→   131→      </div>
   132→   132→    </div>
   133→   133→  )
   134→   134→}
