   159→   159→function calculateAge(dateOfBirth: string | null): string {
   160→   160→  if (!dateOfBirth) return 'N/A'
   161→   161→  const dob = new Date(dateOfBirth)
   162→   162→  const today = new Date()
   163→   163→  const age = today.getFullYear() - dob.getFullYear()
   164→   164→  const monthDiff = today.getMonth() - dob.getMonth()
   165→   165→  if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < dob.getDate())) {
   166→   166→    return `${age - 1} years`
   167→   167→  }
   168→   168→  return `${age} years`
   169→   169→}

   187→   187→// QR-like visual component for patient ID
   188→   188→function PatientQRCode({ patientId }: { patientId: string }) {
   189→   189→  // Generate a simple visual representation
   190→   190→  const hash = patientId.split('').reduce((acc, char) => acc + char.charCodeAt(0), 0)
   191→   191→  const cells: boolean[][] = []
   192→   192→  for (let i = 0; i < 11; i++) {
   193→   193→    const row: boolean[] = []
   194→   194→    for (let j = 0; j < 11; j++) {
   195→   195→      row.push(((i * 11 + j + hash) * 7) % 3 !== 0)
   196→   196→    }
   197→   197→    cells.push(row)
   198→   198→  }
   199→   199→
   200→   200→  return (
   201→   201→    <div className="flex flex-col items-center gap-3">
   202→   202→      <div className="border-4 border-green-700 p-2 bg-white rounded-lg">
   203→   203→        <div className="grid grid-cols-11 gap-px">
   204→   204→          {cells.flat().map((filled, i) => (
   205→   205→            <div
   206→   206→              key={i}
   207→   207→              className={`w-3 h-3 ${filled ? 'bg-green-950' : 'bg-white'}`}
   208→   208→            />
   209→   209→          ))}
   210→   210→        </div>
   211→   211→      </div>
   212→   212→      <p className="text-sm font-mono font-bold text-green-900">{patientId}</p>
   213→   213→      <p className="text-xs text-muted-foreground">Patient ID</p>
   214→   214→    </div>
   215→   215→  )
   216→   216→}
