    87→function getPOStatusBadge(status: string) {
    88→  const map: Record<string, { label: string; className: string }> = {
    89→    PENDING: { label: "Pending", className: "bg-yellow-100 text-yellow-800 border-yellow-200" },
    90→    APPROVED: { label: "Approved", className: "bg-blue-100 text-blue-800 border-blue-200" },
    91→    ORDERED: { label: "Ordered", className: "bg-orange-100 text-orange-800 border-orange-200" },
    92→    DELIVERED: { label: "Delivered", className: "bg-green-100 text-green-800 border-green-200" },
    93→    CANCELLED: { label: "Cancelled", className: "bg-red-100 text-red-800 border-red-200" },
    94→  }
    95→  const s = map[status] || { label: status, className: "" }
    96→  return <Badge variant="outline" className={s.className}>{s.label}</Badge>

   111→function StockLevelBar({ quantity, minStock }: { quantity: number; minStock: number }) {
   112→  const percentage = Math.min((quantity / (minStock * 2)) * 100, 100)
   113→  const isLow = quantity < minStock
   114→  const isCritical = quantity === 0
   115→
   116→  return (
   117→    <div className="w-full">
   118→      <div className="h-2 w-full bg-muted rounded-full overflow-hidden">
   119→        <div
   120→          className={`h-full rounded-full transition-all ${
   121→            isCritical ? "bg-red-500" : isLow ? "bg-orange-400" : "bg-green-600"
   122→          }`}
   123→          style={{ width: `${percentage}%` }}
   124→        />
   125→      </div>
   126→      <div className="flex justify-between mt-1">
   127→        <span className={`text-xs ${isCritical ? "text-red-600 font-bold" : isLow ? "text-orange-600 font-medium" : "text-green-700"}`}>
   128→          {quantity} {isLow ? "(Low!)" : ""}
   129→        </span>
   130→        <span className="text-xs text-muted-foreground">Min: {minStock}</span>
   131→      </div>
   132→    </div>
   133→  )
   134→}
