   189→function getStatusBadge(status: string) {
   190→  switch (status) {
   191→    case 'COMPLETED':
   192→      return <Badge className="bg-green-100 text-green-800 hover:bg-green-100 border-green-200">Completed</Badge>
   193→    case 'CONFIRMED':
   194→      return <Badge className="bg-violet-100 text-violet-700 hover:bg-violet-100 border-violet-200">Confirmed</Badge>
   195→    case 'BOOKED':
   196→      return <Badge className="bg-amber-100 text-amber-700 hover:bg-amber-100 border-amber-200">Booked</Badge>
   197→    case 'CANCELLED':
   198→      return <Badge className="bg-rose-100 text-rose-700 hover:bg-rose-100 border-rose-200">Cancelled</Badge>
   199→    case 'NO_SHOW':
   200→      return <Badge className="bg-gray-100 text-gray-600 hover:bg-gray-100 border-gray-200">No Show</Badge>
   201→    default:
   202→      return <Badge variant="secondary">{status}</Badge>
   203→  }
   204→}
   205→
   206→function getAttendanceBadge(status: string) {
   207→  switch (status) {
   208→    case 'PRESENT':
   209→      return <Badge className="bg-green-100 text-green-800 hover:bg-green-100 border-green-200">Present</Badge>
   210→    case 'ABSENT':
   211→      return <Badge className="bg-rose-100 text-rose-700 hover:bg-rose-100 border-rose-200">Absent</Badge>
   212→    case 'LATE':
   213→      return <Badge className="bg-amber-100 text-amber-700 hover:bg-amber-100 border-amber-200">Late</Badge>
   214→    case 'LEAVE':
   215→      return <Badge className="bg-blue-100 text-blue-700 hover:bg-blue-100 border-blue-200">On Leave</Badge>
   216→    case 'HALF_DAY':
   217→      return <Badge className="bg-orange-100 text-orange-700 hover:bg-orange-100 border-orange-200">Half Day</Badge>
   218→    default:
   219→      return <Badge variant="secondary">{status}</Badge>
   220→  }
   221→}
