   337→function formatBDT(amount: number): string {
   338→  if (amount >= 100000) return `৳${(amount / 100000).toFixed(1)}L`
   339→  return `৳${amount.toLocaleString('en-BD')}`
   340→}
   341→
   342→function formatBDTFull(amount: number): string {
   343→  return `৳${amount.toLocaleString('en-BD')}`
