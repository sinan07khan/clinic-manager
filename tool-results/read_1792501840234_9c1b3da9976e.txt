   106→function getStatusColor(status: string) {
   107→  switch (status) {
   108→    case 'BOOKED': return 'bg-blue-100 text-blue-800 border-blue-200'
   109→    case 'CONFIRMED': return 'bg-yellow-100 text-yellow-800 border-yellow-200'
   110→    case 'COMPLETED': return 'bg-green-100 text-green-900 border-green-200'
   111→    case 'CANCELLED': return 'bg-red-100 text-red-800 border-red-200'
   112→    case 'NO_SHOW': return 'bg-gray-100 text-gray-800 border-gray-200'
   113→    default: return 'bg-gray-100 text-gray-700 border-gray-200'
   114→  }
   115→}

   126→
   127→  // Filters

   129→  const [statusFilter, setStatusFilter] = useState('')
   130→  const [typeFilter, setTypeFilter] = useState('')

   132→  // Dialog states
   133→  const [formOpen, setFormOpen] = useState(false)
   134→  const [editingAppointment, setEditingAppointment] = useState<AppointmentData | null>(null)

   139→  const [form, setForm] = useState({
   140→    patientId: '',
   141→    doctorId: '',
   142→    date: '',
   143→    timeSlot: '',
   144→    type: 'CONSULTATION',
   145→    notes: '',
   146→  })
   147→  const [formSubmitting, setFormSubmitting] = useState(false)
   148→  const [patientSearch, setPatientSearch] = useState('')

   153→    setLoading(true)
   154→    try {
   155→      const params = new URLSearchParams()
   156→      params.set('date', format(selectedDate, 'yyyy-MM-dd'))

   166→        toast({ title: 'Error', description: data.error, variant: 'destructive' })

   184→        const data = await res.json()

   187→        }

   190→      }
   191→    }

   200→        doctorMap.set(apt.doctor.id, apt.doctor)
   201→      }
   202→    })
   203→    setDoctors(Array.from(doctorMap.values()))
   204→  }, [appointments])
   205→
   206→  // Also fetch all doctors by looking at all appointments (no date filter)
   207→  useEffect(() => {
   208→    const fetchAllDoctors = async () => {
   209→      try {
   210→        const res = await fetch('/api/appointments?limit=1000')
   211→        const data = await res.json()
   212→        if (res.ok) {
   213→          const doctorMap = new Map<string, Doctor>()
   214→          data.forEach((apt: AppointmentData) => {
   215→            if (!doctorMap.has(apt.doctor.id)) {
   216→              doctorMap.set(apt.doctor.id, apt.doctor)

   244→      type: 'CONSULTATION',
   245→      notes: '',

   248→    setPatientSearch('')

   287→        headers: { 'Content-Type': 'application/json' },

   323→        toast({ title: 'Error', description: 'Failed to update status', variant: 'destructive' })

   326→      toast({ title: 'Error', description: 'Failed to update status', variant: 'destructive' })
   327→    }
   328→  }

   337→        toast({ title: 'Error', description: 'Failed to delete', variant: 'destructive' })

   340→      toast({ title: 'Error', description: 'Failed to delete appointment', variant: 'destructive' })

   352→

   357→    const availableSlots = 16 - bookedSlots

   365→
   366→  // Filter patients for search
   367→  const filteredPatients = patients.filter(
   368→    (p) =>

   378→    COMPLETED: appointments.filter((a) => a.status === 'COMPLETED').length,
   379→    CANCELLED: appointments.filter((a) => a.status === 'CANCELLED').length,
   380→    NO_SHOW: appointments.filter((a) => a.status === 'NO_SHOW').length,
   381→  }
   382→
   383→  return (
   384→    <div className="space-y-6">
   385→      {/* Header */}
   386→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   387→        <div>
   388→          <h1 className="text-2xl font-bold tracking-tight text-green-950">Appointment Management</h1>
   389→          <p className="text-muted-foreground">Schedule and manage patient appointments</p>
   390→        </div>
   391→        <div className="flex gap-2">
   392→          <Button
   393→            variant="outline"
   394→            onClick={() => setShowDoctorScheduleDialog(true)}
   395→            className="text-green-800 border-green-200"
   396→          >
   397→            <CalendarDays className="mr-2 h-4 w-4" />
   398→            Doctor Schedule
   399→          </Button>
   400→          <Button onClick={openAddDialog} className="bg-green-700 hover:bg-green-800 text-white">
   401→            <Plus className="mr-2 h-4 w-4" />
   402→            New Appointment
   403→          </Button>
   404→        </div>
   405→      </div>
   406→
   407→      {/* Date Navigation */}
   408→      <Card className="border-green-200 bg-gradient-to-r from-green-50 to-green-50">
   409→        <CardContent className="p-4">
   410→          <div className="flex items-center justify-between">
   411→            <Button variant="outline" size="sm" onClick={() => navigateDate('prev')}>
   412→              <ChevronLeft className="h-4 w-4 mr-1" />
   413→              Previous
   414→            </Button>
   415→            <div className="text-center">
   416→              <h2 className="text-xl font-bold text-green-950">
   417→                {format(selectedDate, 'EEEE')}
   418→              </h2>
   419→              <p className="text-sm text-muted-foreground">
   420→                {format(selectedDate, 'MMMM d, yyyy')}
   421→              </p>
   422→              {isToday(selectedDate) && (
   423→                <Badge className="mt-1 bg-green-100 text-green-900 border-green-200">
   424→                  Today
   425→                </Badge>
   426→              )}
   427→            </div>
   428→            <Button variant="outline" size="sm" onClick={() => navigateDate('next')}>
   429→              Next
   430→              <ChevronRight className="h-4 w-4 ml-1" />
   431→            </Button>
   432→          </div>
   433→          <div className="flex justify-center mt-2">
   434→            <Button
   435→              variant="ghost"
   436→              size="sm"
   437→              onClick={() => navigateDate('today')}
   438→              className="text-green-800 hover:text-green-950"
   439→            >
   440→              <Calendar className="h-4 w-4 mr-1" />
   441→              Go to Today
   442→            </Button>
   443→          </div>
   444→        </CardContent>
   445→      </Card>
   446→
   447→      {/* Status Summary Cards */}
   448→      <div className="grid grid-cols-2 sm:grid-cols-5 gap-3">
   449→        {STATUS_OPTIONS.map((status) => (
   450→          <Card
   451→            key={status}
   452→            className={`cursor-pointer transition-colors ${

   461→              <Badge className={`mt-1 text-xs ${getStatusColor(status)}`}>
   462→                {status.replace('_', ' ')}
   463→              </Badge>
   464→            </CardContent>

   467→      </div>
   468→
   469→      <div className="flex flex-col lg:flex-row gap-6">
   470→        {/* Main Content */}
   471→        <div className="flex-1 space-y-4">
   472→          {/* Filter Bar */}
   473→          <Card>

   475→              <div className="flex flex-col sm:flex-row gap-3">

   477→                  <Select value={doctorFilter} onValueChange={(v) => setDoctorFilter(v === 'ALL' ? '' : v)}>

   480→                      <SelectValue placeholder="All Doctors" />
   481→                    </SelectTrigger>
   482→                    <SelectContent>
   483→                      <SelectItem value="ALL">All Doctors</SelectItem>
   484→                      {doctors.map((doc) => (
   485→                        <SelectItem key={doc.id} value={doc.id}>
   486→                          {doc.name}
   487→                        </SelectItem>
   488→                      ))}
   489→                    </SelectContent>
   490→                  </Select>

   492→                <Select value={typeFilter} onValueChange={(v) => setTypeFilter(v === 'ALL' ? '' : v)}>

   494→                    <Filter className="mr-2 h-4 w-4" />
   495→                    <SelectValue placeholder="All Types" />
   496→                  </SelectTrigger>
   497→                  <SelectContent>

   506→                    variant="ghost"
   507→                    size="sm"
   508→                    onClick={() => setStatusFilter('')}
   509→                    className="text-muted-foreground"
   510→                  >
   511→                    <XCircle className="h-4 w-4 mr-1" />
   512→                    Clear filter
   513→                  </Button>

   515→              </div>
   516→            </CardContent>
   517→          </Card>
   518→
   519→          {/* Appointments Table */}
   520→          <Card>
   521→            <CardHeader className="pb-3">
   522→              <CardTitle className="text-lg">
   523→                {format(selectedDate, 'MMM d')} Appointments
   524→              </CardTitle>
   525→              <CardDescription>
   526→                {appointments.length} appointment{appointments.length !== 1 ? 's' : ''} scheduled
   527→              </CardDescription>
   528→            </CardHeader>
   529→            <CardContent className="p-0">
   530→              <div className="overflow-x-auto">
   531→                <Table>
   532→                  <TableHeader>
   533→                    <TableRow className="bg-muted/50">
   534→                      <TableHead className="w-[100px]">Time</TableHead>
   535→                      <TableHead>Patient</TableHead>
   536→                      <TableHead className="hidden md:table-cell">Doctor</TableHead>
   537→                      <TableHead className="hidden sm:table-cell">Type</TableHead>
   538→                      <TableHead>Status</TableHead>
   539→                      <TableHead className="text-right">Actions</TableHead>
   540→                    </TableRow>
   541→                  </TableHeader>
   542→                  <TableBody>
   543→                    {loading ? (

   548→                      </TableRow>
   549→                    ) : appointments.length === 0 ? (
   550→                      <TableRow>
   551→                        <TableCell colSpan={6} className="text-center py-12 text-muted-foreground">
   552→                          <Calendar className="h-12 w-12 mx-auto mb-3 opacity-30" />
   553→                          No appointments for this date.
   554→                        </TableCell>
   555→                      </TableRow>
   556→                    ) : (
   557→                      appointments.map((apt) => (
   558→                        <TableRow key={apt.id} className="hover:bg-muted/50 transition-colors">
   559→                          <TableCell className="font-mono text-sm">
   560→                            <div className="flex items-center gap-1">
   561→                              <Clock className="h-3 w-3 text-muted-foreground" />
   562→                              {apt.timeSlot}
   563→                            </div>
   564→                          </TableCell>
   565→                          <TableCell>
   566→                            <div>
   567→                              <p className="font-medium text-sm">{apt.patient.name}</p>
   568→                              <p className="text-xs text-muted-foreground">
   569→                                {apt.patient.patientId} &middot; {apt.patient.phone}
   570→                              </p>
   571→                            </div>
   572→                          </TableCell>
   573→                          <TableCell className="hidden md:table-cell">
   574→                            <div>
   575→                              <p className="text-sm font-medium">{apt.doctor.name}</p>
   576→                              {apt.doctor.specialization && (
   577→                                <p className="text-xs text-muted-foreground">{apt.doctor.specialization}</p>

   579→                            </div>
   580→                          </TableCell>
   581→                          <TableCell className="hidden sm:table-cell">
   582→                            <Badge variant="outline" className="text-xs">
   583→                              {apt.type.replace('_', ' ')}
   584→                            </Badge>
