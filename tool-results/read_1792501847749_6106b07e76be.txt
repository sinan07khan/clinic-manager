   189→   189→function getStatusBadge(status: string) {
   190→   190→  switch (status) {
   191→   191→    case 'COMPLETED':
   192→   192→      return <Badge className="bg-green-100 text-green-800 hover:bg-green-100 border-green-200">Completed</Badge>
   193→   193→    case 'CONFIRMED':
   194→   194→      return <Badge className="bg-violet-100 text-violet-700 hover:bg-violet-100 border-violet-200">Confirmed</Badge>
   195→   195→    case 'BOOKED':
   196→   196→      return <Badge className="bg-amber-100 text-amber-700 hover:bg-amber-100 border-amber-200">Booked</Badge>
   197→   197→    case 'CANCELLED':
   198→   198→      return <Badge className="bg-rose-100 text-rose-700 hover:bg-rose-100 border-rose-200">Cancelled</Badge>
   199→   199→    case 'NO_SHOW':
   200→   200→      return <Badge className="bg-gray-100 text-gray-600 hover:bg-gray-100 border-gray-200">No Show</Badge>
   201→   201→    default:
   202→   202→      return <Badge variant="secondary">{status}</Badge>
   203→   203→  }
   204→   204→}
