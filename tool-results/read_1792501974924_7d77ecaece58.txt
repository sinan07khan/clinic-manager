   267→function StaffDirectory() {
   268→  const [staff, setStaff] = useState<StaffMember[]>([])
   269→  const [loading, setLoading] = useState(true)
   270→  const [search, setSearch] = useState("")
   271→  const [filterRole, setFilterRole] = useState("")
   272→  const [filterDept, setFilterDept] = useState("")
   273→  const [filterActive, setFilterActive] = useState("")
   274→
   275→  // Dialog states
   276→  const [showAddDialog, setShowAddDialog] = useState(false)
   277→  const [editingStaff, setEditingStaff] = useState<StaffMember | null>(null)
   278→
   279→  // Form state
   280→  const [formName, setFormName] = useState("")
   281→  const [formEmail, setFormEmail] = useState("")
   282→  const [formPhone, setFormPhone] = useState("")
   283→  const [formRole, setFormRole] = useState("STAFF")
   284→  const [formDept, setFormDept] = useState("")
   285→  const [formSpec, setFormSpec] = useState("")
   286→  const [formActive, setFormActive] = useState(true)
   287→  const [submitting, setSubmitting] = useState(false)
   288→
   289→  const fetchStaff = useCallback(async () => {
   290→    setLoading(true)
   291→    try {
   292→      const params = new URLSearchParams()
   293→      if (search) params.set("search", search)
   294→      if (filterRole) params.set("role", filterRole)
   295→      if (filterDept) params.set("department", filterDept)
   296→      if (filterActive) params.set("isActive", filterActive)
   297→
   298→      const res = await fetch(`/api/staff?${params.toString()}`)
   299→      if (res.ok) {
   300→        const data = await res.json()
   301→        setStaff(data)
   302→      }
   303→    } catch (err) {
   304→      console.error(err)
   305→    } finally {
   306→      setLoading(false)
   307→    }
   308→  }, [search, filterRole, filterDept, filterActive])
   309→
   310→  useEffect(() => {
   311→    fetchStaff()
   312→  }, [fetchStaff])
   313→
   314→  const resetForm = () => {
   315→    setFormName("")
   316→    setFormEmail("")
   317→    setFormPhone("")
   318→    setFormRole("STAFF")
   319→    setFormDept("")
   320→    setFormSpec("")
   321→    setFormActive(true)
   322→    setEditingStaff(null)
   323→  }
   324→
   325→  const openAdd = () => {
   326→    resetForm()
   327→    setShowAddDialog(true)
   328→  }
   329→
   330→  const openEdit = (s: StaffMember) => {
   331→    setEditingStaff(s)
   332→    setFormName(s.name)
   333→    setFormEmail(s.email)
   334→    setFormPhone(s.phone)
   335→    setFormRole(s.role)
   336→    setFormDept(s.department || "")
   337→    setFormSpec(s.specialization || "")
   338→    setFormActive(s.isActive)
   339→    setShowAddDialog(true)
   340→  }
   341→
   342→  const handleSubmit = async () => {
   343→    if (!formName || !formEmail || !formPhone) {
   344→      toast.error("Name, email, and phone are required")
   345→      return
   346→    }
   347→    setSubmitting(true)
   348→    try {
   349→      const url = editingStaff ? `/api/staff/${editingStaff.id}` : "/api/staff"
   350→      const method = editingStaff ? "PUT" : "POST"
   351→      const body = {
   352→        name: formName,
   353→        email: formEmail,
   354→        phone: formPhone,
   355→        role: formRole,
   356→        department: formDept || null,
   357→        specialization: formSpec || null,
   358→        isActive: formActive,
   359→      }
   360→
   361→      const res = await fetch(url, {
   362→        method,
   363→        headers: { "Content-Type": "application/json" },
   364→        body: JSON.stringify(body),
   365→      })
   366→
   367→      if (res.ok) {
   368→        toast.success(editingStaff ? "Staff updated successfully" : "Staff added successfully")
   369→        setShowAddDialog(false)
   370→        resetForm()
   371→        fetchStaff()
   372→      } else {
   373→        const data = await res.json()
   374→        toast.error(data.error || "Operation failed")
   375→      }
   376→    } catch (err) {
   377→      toast.error("An error occurred")
   378→    } finally {
   379→      setSubmitting(false)
   380→    }
   381→  }
   382→
   383→  const handleToggleActive = async (s: StaffMember) => {
   384→    try {
   385→      const res = await fetch(`/api/staff/${s.id}`, {
   386→        method: "PUT",
   387→        headers: { "Content-Type": "application/json" },
   388→        body: JSON.stringify({ isActive: !s.isActive }),
   389→      })
   390→      if (res.ok) {
   391→        toast.success(`Staff ${s.isActive ? "deactivated" : "activated"}`)
   392→        fetchStaff()
   393→      }
   394→    } catch (err) {
   395→      toast.error("Failed to update staff status")
   396→    }
   397→  }
   398→
   399→  const handleDelete = async (s: StaffMember) => {
   400→    if (!confirm(`Are you sure you want to deactivate ${s.name}?`)) return
   401→    try {
   402→      const res = await fetch(`/api/staff/${s.id}`, { method: "DELETE" })
   403→      if (res.ok) {
   404→        toast.success("Staff deactivated")
   405→        fetchStaff()
   406→      }
   407→    } catch (err) {
   408→      toast.error("Failed to deactivate staff")
   409→    }
   410→  }
   411→
   412→  // Role count summary
   413→  const roleCounts = staff.reduce<Record<string, number>>((acc, s) => {
   414→    acc[s.role] = (acc[s.role] || 0) + 1
   415→    return acc
   416→  }, {})
   417→
   418→  const activeCount = staff.filter((s) => s.isActive).length
   419→  const inactiveCount = staff.length - activeCount
   420→
   421→  return (
   422→    <div className="space-y-6">
   423→      {/* Summary Cards */}
   424→      <div className="grid grid-cols-2 gap-4 sm:grid-cols-3 lg:grid-cols-5">
   425→        <Card className="border-green-200 bg-green-50/50">
   426→          <CardContent className="p-4">
   427→            <div className="flex items-center gap-2">
   428→              <Users className="h-4 w-4 text-green-700" />
   429→              <span className="text-sm text-green-800">Total Staff</span>
   430→            </div>
   431→            <p className="mt-1 text-2xl font-bold text-green-950">{staff.length}</p>
   432→          </CardContent>
   433→        </Card>
   434→        <Card className="border-green-200 bg-green-50/50">
   435→          <CardContent className="p-4">
   436→            <div className="flex items-center gap-2">
   437→              <UserCheck className="h-4 w-4 text-green-600" />
   438→              <span className="text-sm text-green-700">Active</span>
   439→            </div>
   440→            <p className="mt-1 text-2xl font-bold text-green-900">{activeCount}</p>
   441→          </CardContent>
   442→        </Card>
   443→        <Card className="border-red-200 bg-red-50/50">
   444→          <CardContent className="p-4">
   445→            <div className="flex items-center gap-2">
   446→              <UserX className="h-4 w-4 text-red-600" />
   447→              <span className="text-sm text-red-700">Inactive</span>
   448→            </div>
   449→            <p className="mt-1 text-2xl font-bold text-red-900">{inactiveCount}</p>
   450→          </CardContent>
   451→        </Card>
   452→        <Card className="border-green-200 bg-green-50/50">
   453→          <CardContent className="p-4">
   454→            <div className="flex items-center gap-2">
   455→              <Stethoscope className="h-4 w-4 text-green-700" />
   456→              <span className="text-sm text-green-800">Doctors</span>
   457→            </div>
   458→            <p className="mt-1 text-2xl font-bold text-green-950">{roleCounts.DOCTOR || 0}</p>
   459→          </CardContent>
   460→        </Card>
   461→        <Card className="border-pink-200 bg-pink-50/50">
   462→          <CardContent className="p-4">
   463→            <div className="flex items-center gap-2">
   464→              <ClipboardList className="h-4 w-4 text-pink-600" />
   465→              <span className="text-sm text-pink-700">Nurses</span>
   466→            </div>
   467→            <p className="mt-1 text-2xl font-bold text-pink-900">{roleCounts.NURSE || 0}</p>
   468→          </CardContent>
   469→        </Card>
   470→      </div>
   471→
   472→      {/* Filters & Actions */}
   473→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   474→        <div className="flex flex-1 flex-col gap-3 sm:flex-row sm:items-center">
   475→          <div className="relative flex-1 sm:max-w-xs">
   476→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   477→            <Input
   478→              placeholder="Search by name..."
   479→              value={search}
   480→              onChange={(e) => setSearch(e.target.value)}
   481→              className="pl-9"
   482→            />
   483→          </div>
   484→          <Select value={filterRole} onValueChange={(v) => setFilterRole(v === "ALL" ? "" : v)}>
   485→            <SelectTrigger className="w-full sm:w-[160px]">
   486→              <SelectValue placeholder="All Roles" />
   487→            </SelectTrigger>
   488→            <SelectContent>
   489→              <SelectItem value="ALL">All Roles</SelectItem>
   490→              {ROLES.map((r) => (
   491→                <SelectItem key={r} value={r}>
   492→                  {ROLE_LABELS[r]}
   493→                </SelectItem>
   494→              ))}
   495→            </SelectContent>
   496→          </Select>
   497→          <Select
   498→            value={filterDept}
   499→            onValueChange={(v) => setFilterDept(v === "ALL" ? "" : v)}
   500→          >
   501→            <SelectTrigger className="w-full sm:w-[180px]">
   502→              <SelectValue placeholder="All Departments" />
   503→            </SelectTrigger>
   504→            <SelectContent>
   505→              <SelectItem value="ALL">All Departments</SelectItem>
   506→              {DEPARTMENTS.map((d) => (
   507→                <SelectItem key={d} value={d}>
   508→                  {d}
   509→                </SelectItem>
   510→              ))}
   511→            </SelectContent>
   512→          </Select>
   513→          <Select
   514→            value={filterActive}
   515→            onValueChange={(v) => setFilterActive(v === "ALL" ? "" : v)}
   516→          >
   517→            <SelectTrigger className="w-full sm:w-[130px]">
   518→              <SelectValue placeholder="Status" />
   519→            </SelectTrigger>
   520→            <SelectContent>
   521→              <SelectItem value="ALL">All</SelectItem>
   522→              <SelectItem value="true">Active</SelectItem>
   523→              <SelectItem value="false">Inactive</SelectItem>
   524→            </SelectContent>
   525→          </Select>
   526→        </div>
   527→        <Button onClick={openAdd} className="bg-green-700 hover:bg-green-800">
   528→          <Plus className="mr-2 h-4 w-4" />
   529→          Add Staff
   530→        </Button>
   531→      </div>
   532→
   533→      {/* Staff Table */}
   534→      <Card>
   535→        <CardContent className="p-0">
   536→          <div className="max-h-[520px] overflow-y-auto overflow-x-auto">
   537→            <Table>
   538→              <TableHeader>
   539→                <TableRow className="bg-muted/50">
   540→                  <TableHead className="whitespace-nowrap font-semibold">Name</TableHead>
   541→                  <TableHead className="whitespace-nowrap font-semibold hidden sm:table-cell">Email</TableHead>
   542→                  <TableHead className="whitespace-nowrap font-semibold hidden md:table-cell">Phone</TableHead>
   543→                  <TableHead className="whitespace-nowrap font-semibold">Role</TableHead>
   544→                  <TableHead className="whitespace-nowrap font-semibold hidden lg:table-cell">Department</TableHead>
   545→                  <TableHead className="whitespace-nowrap font-semibold hidden sm:table-cell">Status</TableHead>
   546→                  <TableHead className="font-semibold text-right">Actions</TableHead>
   547→                </TableRow>
   548→              </TableHeader>
   549→              <TableBody>
   550→                {loading ? (
   551→                  <TableRow>
   552→                    <TableCell colSpan={7} className="py-12 text-center text-muted-foreground">
   553→                      Loading...
   554→                    </TableCell>
   555→                  </TableRow>
   556→                ) : staff.length === 0 ? (
   557→                  <TableRow>
   558→                    <TableCell colSpan={7} className="py-12 text-center text-muted-foreground">
   559→                      No staff found
   560→                    </TableCell>
   561→                  </TableRow>
   562→                ) : (
   563→                  staff.map((s) => (
   564→                    <TableRow key={s.id}>
   565→                      <TableCell className="font-medium">{s.name}</TableCell>
   566→                      <TableCell className="max-w-[200px] truncate text-sm text-muted-foreground hidden sm:table-cell">
   567→                        {s.email}
   568→                      </TableCell>
   569→                      <TableCell className="text-sm hidden md:table-cell">{s.phone}</TableCell>
   570→                      <TableCell>
   571→                        <RoleBadge role={s.role} />
   572→                      </TableCell>
   573→                      <TableCell className="text-sm hidden lg:table-cell">{s.department || "—"}</TableCell>
   574→                      <TableCell className="hidden sm:table-cell">
   575→                        <Badge
   576→                          variant="outline"
   577→                          className={
   578→                            s.isActive
   579→                              ? "bg-green-100 text-green-800 border-green-200"
   580→                              : "bg-red-100 text-red-800 border-red-200"
   581→                          }
   582→                        >
   583→                          {s.isActive ? "Active" : "Inactive"}
   584→                        </Badge>
   585→                      </TableCell>
   586→                      <TableCell className="text-right">
   587→                        <DropdownMenu>
