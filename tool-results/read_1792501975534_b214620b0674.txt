   110→function formatTime12h(time24: string) {
   111→  if (!time24) return ''
   112→  const [h, m] = time24.split(':').map(Number)
   113→  const ampm = h >= 12 ? 'PM' : 'AM'
   114→  const hour12 = h % 12 || 12
   115→  return `${hour12}:${String(m).padStart(2, '0')} ${ampm}`
   116→}
   117→
   118→function calculateSlots(startTime: string, endTime: string, slotDuration: number, breakStart?: string | null, breakEnd?: string | null): number {
   119→  const toMinutes = (t: string) => {
   120→    const [h, m] = t.split(':').map(Number)
   121→    return h * 60 + m
   122→  }
   123→
   124→  let totalMinutes = toMinutes(endTime) - toMinutes(startTime)
   125→  if (totalMinutes <= 0) return 0
   126→
   127→  if (breakStart && breakEnd) {
   128→    const breakMinutes = toMinutes(breakEnd) - toMinutes(breakStart)
   129→    if (breakMinutes > 0) totalMinutes -= breakMinutes
   130→  }
   131→
   132→  return Math.max(0, Math.floor(totalMinutes / slotDuration))
   133→}
