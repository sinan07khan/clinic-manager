   159→function getStatusBadge(status: string) {
   160→  const map: Record<string, { label: string; className: string }> = {
   161→    PENDING: { label: "Pending", className: "bg-yellow-100 text-yellow-800 border-yellow-200" },
   162→    PROCESSING: { label: "Processing", className: "bg-blue-100 text-blue-800 border-blue-200" },
   163→    COMPLETED: { label: "Completed", className: "bg-green-100 text-green-800 border-green-200" },
   164→    REVIEWED: { label: "Reviewed", className: "bg-green-100 text-green-900 border-green-200" },
   165→    COLLECTED: { label: "Collected", className: "bg-blue-100 text-blue-800 border-blue-200" },
   166→    TESTING: { label: "Testing", className: "bg-yellow-100 text-yellow-800 border-yellow-200" },
   167→    DISCARDED: { label: "Discarded", className: "bg-red-100 text-red-800 border-red-200" },
   168→  }
   169→  const s = map[status] || { label: status, className: "" }
   170→  return <Badge variant="outline" className={s.className}>{s.label}</Badge>
   171→}

   353→        if (ref.startsWith("<") || ref.startsWith(">")) {
   354→          const threshold = parseFloat(ref.replace(/[<>]/g, ""))
   355→          updated[index].isAbnormal = !isNaN(threshold) && (
   356→            (ref.startsWith("<") && val >= threshold) || (ref.startsWith(">") && val <= threshold)
   357→          )
   358→        } else if (ref.includes("-")) {
   359→          const [low, high] = ref.split("-").map(Number)
   360→          updated[index].isAbnormal = !isNaN(low) && !isNaN(high) && (val < low || val > high)

   362→          updated[index].isAbnormal = value !== "" && value !== "Negative" && value !== "negative"
